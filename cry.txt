# detection-test.ps1 — als Administrator ausfuehren
# Simuliert mehrere unabhaengige TTP-aehnliche Aktionen zum Testen der Wazuh-Erkennung.

if (-NOT ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Bitte als Administrator ausfuehren." -ForegroundColor Red
    exit
}

Write-Host "--- Aktion 1: Datei mit EICAR-Testsignatur in Temp ablegen ---"
$eicar = 'X5O!P%@AP[4\PZX54(P^)7CC)7}$EICAR-STANDARD-ANTIVIRUS-TEST-FILE!$H+H*'
Set-Content -Path "$env:TEMP\test-payload.txt" -Value $eicar -Encoding ASCII

Write-Host "--- Aktion 2: Neuer lokaler Benutzer ---"
$pw = ConvertTo-SecureString "Tmp#2026!xQ9" -AsPlainText -Force
New-LocalUser -Name "svc_update" -Password $pw -FullName "Update Service" -Description "temp test account" -ErrorAction SilentlyContinue

Write-Host "--- Aktion 3: Registry Run-Key (Autostart) ---"
New-ItemProperty -Path "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run" -Name "SysHelper" -Value "$env:TEMP\test-payload.txt" -PropertyType String -Force

Write-Host "--- Aktion 4: Geplante Aufgabe ---"
$action = New-ScheduledTaskAction -Execute "cmd.exe" -Argument "/c echo test"
$trigger = New-ScheduledTaskTrigger -Once -At (Get-Date).AddDays(1)
Register-ScheduledTask -TaskName "SystemCheckTask" -Action $action -Trigger $trigger -Force

Write-Host "--- Aktion 5: Windows Defender Echtzeitschutz per Registry deaktivieren ---"
New-ItemProperty -Path "HKLM:\SOFTWARE\Policies\Microsoft\Windows Defender" -Name "DisableAntiSpyware" -Value 1 -PropertyType DWORD -Force

Write-Host "--- Aktion 6: Firewall-Regel hinzufuegen ---"
New-NetFirewallRule -DisplayName "TempAllow4444" -Direction Inbound -LocalPort 4444 -Protocol TCP -Action Allow

Write-Host "--- Aktion 7: Application-Eventlog leeren ---"
wevtutil cl Application

Write-Host "--- Fertig. 7 Aktionen ausgefuehrt. ---"
